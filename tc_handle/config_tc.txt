# TC-based Multi-WAN Load Balancer Configuration
#
# Format:
#   local <interface>                    - Local interface (facing clients)
#   remote <network/prefix>              - Remote network to load balance
#   wan <interface> <gateway> [peer_ip]  - WAN interface configuration
#   method <roundrobin|hash>             - Load balancing method
#
# Workflow:
#   Packets from LOCAL destined to REMOTE network will be distributed
#   across WAN interfaces using the specified method.
#
#   CLIENT (192.168.1.x) --> LOCAL (eth0) --> KERNEL (TC + multipath) --> WAN1/WAN2/WAN3 --> SERVER (192.168.2.x)

# Local interface - receives traffic from clients
local eth0

# Remote network - destination network to load balance
remote 192.168.2.0/24

# WAN interfaces - each line defines one WAN path
# Format: wan <interface> <gateway_ip> [health_check_ip]
#
# gateway_ip: Next-hop gateway for this WAN
# health_check_ip: IP to ping for health checks (optional, defaults to gateway)
wan eth1 10.0.1.1 8.8.8.8
wan eth2 10.0.2.1 8.8.8.8
wan eth3 10.0.3.1 8.8.8.8

# Load balancing method
# - roundrobin: Distribute packets across WANs in round-robin fashion
# - hash: Hash by flow (src_ip + dst_ip + src_port + dst_port) for session affinity
method roundrobin
